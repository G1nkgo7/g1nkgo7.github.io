void solve() {
int n = next(), idx = 0;
int son[100000 * 32 + 10][2];

auto insert = [&](int x) {
    int p = 0;
    for (int i = 31; i >= 0; --i) {
        int c = (x >> i) & 1;
        if (!son[p][c])
            son[p][c] = ++idx;
        p = son[p][c];
    }
};
auto query = [&](int x) {
    int p = 0, res = 0;
    for (int i = 31; i >= 0; --i) {
        int c = (x >> i) & 1;
        int rev = 1 - c;
        if (son[p][rev]) {
            res |= (1 << i);
            p = son[p][rev];
        } else
            p = son[p][c];
    }
    return res;
};

int ans = 0;
insert(next<int>());
for (int i = 1; i < n ; ++i) {
    int x = next<int>();
    ans = max(ans, query(x));
    insert(x);
}
cout << ans;

}
